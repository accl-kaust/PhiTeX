% \filemoddate of a file the job is writing (its transcript, which
% hyperxmp dates under XeTeX for want of \pdfcreationdate; an \openout
% file): a date, not nothing. A real run reads its time from the disk,
% so only the date's shape is compared: its start, length and zone.
\documentclass{article}
\def\cnt#1{\ifx#1\relax\else\advance\count255 1 \expandafter\cnt\fi}
\def\shape#1{\count255=0 \expandafter\cnt#1\relax
  \expandafter\first#1\end{} (\the\count255\ characters)}
\def\first#1#2#3\end{#1#2\last#3\end}
\def\last#1#2\end{\ifx\end#2\end ending #1\else\last#2\end\fi}
\edef\logdate{\filemoddate{\jobname.log}}
\immediate\openout1=\jobname-w.txt
\immediate\write1{written}
\edef\writedate{\filemoddate{\jobname-w.txt}}
\begin{document}
The transcript's date: \shape\logdate.

The open file's date: \shape\writedate.
\end{document}
