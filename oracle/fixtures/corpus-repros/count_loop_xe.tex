% !TEX program = xelatex
% Loops of register arithmetic and code assignments, in the shapes a
% command loop can make without reading the tokens, and the shapes it
% must leave to the interpreter; XeTeX: the codes past 255.
\documentclass{article}
\makeatletter
\newcount\ca \newcount\cb \newcount\cc
\newcommand\sect[1]{\texttt{#1}: }
\def\lc#1{\the\lccode#1\space}
\begin{document}

\sect{lc in group}
\begingroup
  \ca=0 \cb=256
  \loop
    \lccode\ca=\ca
    \advance\ca by 1
  \ifnum\ca<\cb
  \repeat
  \lccode`\_=`\ \relax
  \lowercase{\xdef\res{Ab_Cd}}%
  \the\ca\ \lc{65} \lc{200} \lc{`\_}
\endgroup
\res\ \the\ca\ \lc{65} \lc{200} \lc{`\_}
\par

\sect{uc, chardef step, chardef limit}
\begingroup
  \count@=64
  \chardef\lim=90
  \loop
    \uccode\count@=\count@
    \advance\count@\@ne
  \ifnum\count@<\lim
  \repeat
  \the\count@\ \the\uccode 70 \ \the\uccode 97
\endgroup
\the\uccode 70 \ \the\uccode 97
\par

\sect{cat, constant value}
\begingroup
  \count@=48

  \the\catcode`\5\ \the\count@
  \def\a1b{one}\a1b
\endgroup
\the\catcode`\5\par
