% A macro-heavy plain TeX job: argument scanning (undelimited and
% delimited), \csname, \edef, \expandafter chains, conditionals,
% registers, groups, token registers, and some typesetting.
\catcode`\@=11
\newcount\n \newcount\m \newcount\k \newtoks\acc \newdimen\d \newcount\j
\def\step#1#2\end{\advance\n by 1
  \expandafter\def\csname x@\number\n\endcsname{#1#2}}
\def\mapit#1{\ifx#1\relax\else\step#1rest of it\end\expandafter\mapit\fi}
\def\alpha@{\mapit abcdefghijklmnopqrstuvwxyz\relax}
\def\fib#1#2#3{\ifnum#3>0 \m=#1 \advance\m by #2
  \j=#3 \advance\j by -1 \edef\next{\noexpand\fib{#2}{\the\m}{\the\j}}%
  \else\let\next\relax\fi\next}
\def\addto#1#2{\expandafter\def\expandafter#1\expandafter{#1#2}}
\def\keyval#1=#2,{\ifx\relax#1\else
  \expandafter\edef\csname kv@#1\endcsname{#2}\expandafter\keyval\fi}
\def\setkeys#1{\keyval#1,\relax=,}
\def\reverse#1{\rev@#1\@nil{}}
\def\rev@#1#2\@nil#3{\ifx\relax#2\relax\def\next{#1#3}\else
  \def\next{\rev@#2\@nil{#1#3}}\fi\next}
\def\words{The quick brown fox jumps over the lazy dog and keeps running
  far beyond the hills while macros expand.}
\def\para{{\setkeys{width=3pt,height=4pt,depth=1pt,color=red,shape=circle}%
  \d=\csname kv@width\endcsname \advance\d by \csname kv@height\endcsname
  \acc={}\k=0
  \loop\ifnum\k<12 \advance\k by 1
    \acc=\expandafter{\the\acc\csname x@\number\k\endcsname\ }\repeat
  \edef\r{\reverse{abcdefghij}}%
  \fib{1}{1}{20}%
  \setbox0\hbox{\the\acc}\r\ \the\m\par}}
\alpha@
\hsize=30pc \vsize=50pc \parindent=1em
\count@=0
\loop\ifnum\count@<40000 \advance\count@ by 1 \para\repeat
\bye
